3) Data Models
type Asset = {
  id: string;
  name: string;
  category: 'Electronics'|'Fashion'|'Watch'|'Other';
  images: string[];
  confidence: number;
  attributes: Record<string, string|number|boolean>;
  condition: 'NEW'|'USED'|'LIKE_NEW';
  country: 'USA'|'TURKEY';
  purchasePrice?: { amount: number; currency: 'USD'|'TRY' };
  marketValue: { amount: number; currency: 'USD'|'TRY' };
  offers: Offer[];
  history: { ts: number; amount: number; currency: 'USD'|'TRY' }[];
  createdAt: number;
  updatedAt: number;
}

type Offer = {
  source: string; // Target, BestBuy, Trendyol...
  url: string;
  amount: number;
  currency: 'USD'|'TRY';
  country: 'USA'|'TURKEY';
  condition: 'NEW'|'USED';
  fetchedAt: number;
  qualityScore: number; // based on trust signals
  isOutlier?: boolean;
}
